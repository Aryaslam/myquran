import { CFG } from './config.js';
import { fetch } from '@tauri-apps/plugin-http';
import { appDataDir, join } from '@tauri-apps/api/path';
import { readDir, exists, mkdir, open, writeTextFile, rename } from '@tauri-apps/plugin-fs';

const pad3 = (n) => String(n).padStart(3, '0');

let readsCache = null;

async function getFolderUrl() {
  if (!readsCache) {
    const res = await fetch(CFG.readsApi);
    const reads = await res.json();
    const r = reads.find(x => x.id === CFG.readId) ||
              reads.find(x => x.name.includes('Alafasy')) ||
              reads.find(x => x.name.includes('العفاسي')) ||
              reads[0];
    readsCache = r;
  }
  return readsCache.folder_url.endsWith('/') ? readsCache.folder_url : readsCache.folder_url + '/';
}

export function remoteUrl(id) {
  return getFolderUrl().then(url => `${url}${pad3(id)}.mp3`);
}

export async function localAudioSrc(id) {
  const base = await appDataDir();
  const path = await join(base, 'surah', pad3(id), 'audio.mp3');
  return `asset://localhost${path}`;
}

export async function scanDownloaded() {
  const base = await appDataDir();
  const surahDir = await join(base, 'surah');
  if (!(await exists(surahDir))) return [];
  const entries = await readDir(surahDir);
  const ids = [];
  for (const e of entries) {
    if (e.isDirectory) {
      const donePath = await join(surahDir, e.name, 'done.json');
      if (await exists(donePath)) {
        ids.push(parseInt(e.name, 10));
      }
    }
  }
  return ids;
}

export async function downloadSurah(id, onProgress) {
  const base = await appDataDir();
  const surahDir = await join(base, 'surah', pad3(id));
  await mkdir(surahDir, { recursive: true });

  // fetch equran
  const equranRes = await fetch(`${CFG.equranApi}/${id}`);
  const equranData = await equranRes.json();
  await writeTextFile(await join(surahDir, 'surat.json'), JSON.stringify(equranData));

  // fetch timing
  const timingRes = await fetch(`${CFG.timingApi}?surah=${id}&read=${CFG.readId}`);
  const timingData = await timingRes.json();
  await writeTextFile(await join(surahDir, 'timing.json'), JSON.stringify(timingData));

  // stream mp3
  const audioUrl = await remoteUrl(id);
  const audioRes = await fetch(audioUrl);
  const total = Number(audioRes.headers.get('content-length') || 0);
  const partPath = await join(surahDir, 'audio.mp3.part');
  const file = await open(partPath, { write: true, create: true, truncate: true });
  const reader = audioRes.body.getReader();
  let received = 0;
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    await file.write(value);
    received += value.length;
    if (total > 0 && onProgress) onProgress(Math.min(100, Math.round(received / total * 100)));
  }
  await file.close();
  const finalPath = await join(surahDir, 'audio.mp3');
  await rename(partPath, finalPath);
  await writeTextFile(await join(surahDir, 'done.json'), JSON.stringify({ id, time: Date.now() }));
}

/* playback */
async function play(i) {
  playToken++;
  const token = playToken;
  S.cur = i;
  const s = SURAH[i];
  const b = buildLines(s);
  S.lines = b.list;
  S.dur = b.dur;
  S.t = 0;
  S.playing = true;
  buildLyrics();
  ui();
  uiProg();

  const downloaded = S.dl.has(s.id) && S.dl.get(s.id).p >= 100;
  let src;
  if (downloaded) {
    src = await localAudioSrc(s.id);
  } else {
    if (S.offline || !navigator.onLine) {
      showError('Tidak ada koneksi. Surah ini belum diunduh.');
      S.playing = false;
      ui();
      return;
    }
    src = await remoteUrl(s.id);
  }

  audio.src = src;
  audio.currentTime = 0;

  const handlePlay = () => {
    if (token !== playToken) return;
    S.playing = true;
    ui();
  };
  const handlePause = () => {
    if (token !== playToken) return;
    S.playing = false;
    ui();
  };
  const handleEnded = () => {
    if (token !== playToken) return;
    next();
  };
  const handleError = () => {
    if (token !== playToken) return;
    const err = audio.error;
    let msg = 'Terjadi kesalahan saat memutar audio.';
    if (err) {
      if (err.code === 4) msg = 'Audio surah ini tidak ditemukan di server.';
      else if (err.code === 2) msg = 'Koneksi terputus. Buka tab Terunduh untuk memutar surah offline.';
    }
    audio.pause();
    S.playing = false;
    ui();
    showError(msg);
  };
  const handleLoadedMeta = () => {
    if (token !== playToken) return;
    S.dur = audio.duration || S.dur;
    const ratio = S.dur / (b.dur || 1);
    S.lines.forEach(l => { l.s *= ratio; l.e *= ratio; });
    uiProg();
  };
  const handleWaiting = () => {
    if (token !== playToken) return;
    $('mPlay').classList.add('buf');
    $('pPlay').classList.add('buf');
    if (!audio._bufTimer) {
      audio._bufTimer = setTimeout(() => {
        if (token !== playToken) return;
        audio.pause();
        S.playing = false;
        ui();
        showError('Koneksi terputus. Buka tab Terunduh untuk memutar surah offline.');
      }, 8000);
    }
  };
  const handleCanPlay = () => {
    if (token !== playToken) return;
    $('mPlay').classList.remove('buf');
    $('pPlay').classList.remove('buf');
    if (audio._bufTimer) { clearTimeout(audio._bufTimer); audio._bufTimer = null; }
  };

  audio.removeEventListener('play', handlePlay);
  audio.removeEventListener('pause', handlePause);
  audio.removeEventListener('ended', handleEnded);
  audio.removeEventListener('error', handleError);
  audio.removeEventListener('loadedmetadata', handleLoadedMeta);
  audio.removeEventListener('durationchange', handleLoadedMeta);
  audio.removeEventListener('waiting', handleWaiting);
  audio.removeEventListener('canplay', handleCanPlay);

  audio.addEventListener('play', handlePlay);
  audio.addEventListener('pause', handlePause);
  audio.addEventListener('ended', handleEnded);
  audio.addEventListener('error', handleError);
  audio.addEventListener('loadedmetadata', handleLoadedMeta);
  audio.addEventListener('durationchange', handleLoadedMeta);
  audio.addEventListener('waiting', handleWaiting);
  audio.addEventListener('canplay', handleCanPlay);

  try {
    await audio.play();
  } catch (e) {
    if (e.name !== 'AbortError') handleError();
  }
}

function initVolSlider(sliderEl, fillEl, thumbEl) {

let toastEl = null;
let lastMsg = '';
let lastTime = 0;
let audioCtx = null;

function ensureToast() {
  if (toastEl) return toastEl;
  toastEl = document.createElement('div');
  toastEl.id = 'toast';
  toastEl.setAttribute('role', 'alert');
  toastEl.style.cssText = `
    position: fixed;
    left: 50%;
    bottom: 104px;
    transform: translateX(-50%) translateY(16px);
    z-index: 80;
    background: var(--ink);
    color: #fff;
    font: 500 13px Inter, system-ui, sans-serif;
    padding: 12px 18px 12px 28px;
    border-radius: 16px;
    max-width: min(460px, calc(100vw - 32px));
    opacity: 0;
    pointer-events: auto;
    box-sizing: border-box;
  `;
  toastEl.innerHTML = `<span style="position:absolute;left:10px;top:50%;margin-top:-4px;width:8px;height:8px;border-radius:50%;background:#d0524a;"></span><span id="toastText"></span>`;
  document.body.appendChild(toastEl);
  toastEl.addEventListener('click', hideToast);
  return toastEl;
}

function hideToast() {
  if (!toastEl) return;
  toastEl.style.transition = 'transform 0.5s var(--ease), opacity 0.5s var(--ease)';
  toastEl.style.transform = 'translateX(-50%) translateY(16px)';
  toastEl.style.opacity = '0';
  setTimeout(() => { if (toastEl) toastEl.remove(); toastEl = null; }, 500);
}

export function showError(message) {
  const now = Date.now();
  if (message === lastMsg && now - lastTime < 3000) return;
  lastMsg = message;
  lastTime = now;
  const el = ensureToast();
  el.querySelector('#toastText').textContent = message;
  el.style.transition = 'transform 0.7s var(--ease), opacity 0.7s var(--ease)';
  el.style.transform = 'translateX(-50%) translateY(0)';
  el.style.opacity = '1';
  setTimeout(hideToast, 4500);
  ding();
}

function ding() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
  if (audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = 'sine';
  osc.frequency.value = 1046;
  gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
  osc.connect(gain).connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + 0.5);
}
